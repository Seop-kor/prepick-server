# 인증 테이블 자동증가 정수 ID 전환

`"user"`, `otp_challenge`, `"session"`의 기본 키와 `session.user_id`를 32비트 정수로 사용한다. GraphQL `User.id`는 계속 `ID!`이며 응답 값은 십진수 문자열이다. JWT의 `sub`와 refresh token의 `sessionId.secret`에서 ID도 십진수 문자열이다. 이전 UUID 토큰은 전환 후 사용할 수 없으므로 사용자는 다시 로그인한다. OTP와 refresh secret 자체는 계속 암호학적으로 무작위 생성한다.

새 테이블을 만들 때의 스키마는 다음과 같다. 이미 UUID 테이블이 있으면 `CREATE TABLE` 대신 아래 전환 SQL을 사용한다.

~~~sql
CREATE TABLE "user" (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name varchar(50) NOT NULL,
  phone varchar(11) NOT NULL CONSTRAINT user_phone_unique UNIQUE,
  password varchar(255) NOT NULL,
  created_at timestamptz NOT NULL,
  updated_at timestamptz NOT NULL
);

CREATE TABLE otp_challenge (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  phone varchar(11) NOT NULL,
  otp varchar(64) NOT NULL,
  expires_at timestamptz NOT NULL,
  attempt_count integer NOT NULL,
  verified_at timestamptz,
  verification_token varchar(64) CONSTRAINT otp_challenge_verification_token_unique UNIQUE,
  verification_expires_at timestamptz,
  consumed_at timestamptz,
  invalidated_at timestamptz,
  created_at timestamptz NOT NULL
);

CREATE TABLE "session" (
  id integer GENERATED BY DEFAULT AS IDENTITY CONSTRAINT refresh_session_pkey PRIMARY KEY,
  user_id integer NOT NULL CONSTRAINT refresh_session_user_id_unique UNIQUE,
  refresh_token varchar(255) NOT NULL,
  expires_at timestamptz NOT NULL,
  created_at timestamptz NOT NULL,
  updated_at timestamptz NOT NULL
);
~~~

## 기존 UUID 데이터 전환

앱을 멈추고 DB를 백업한 뒤 아래 SQL을 수동으로 실행한다. 트랜잭션은 기존 사용자·OTP·세션 행과 세션의 사용자 연결을 보존한다. 기존 UUID는 `legacy_*` 컬럼에 남아 롤백에 사용된다. 실제 DB의 인증 테이블은 설계 시점에 모두 비어 있었지만, 실행 전 행이 생겨도 이 SQL은 매핑한다. 매장 데이터에는 별도의 [매장 ID 전환 SQL](2026-09-25-consumer-store-discovery-design.md#기존-uuid-데이터-전환)을 적용한다. 두 스크립트를 모두 실행해야 현재 DB의 UUID 컬럼이 기본 키·연결 키에서 제거된다.

~~~sql
BEGIN;
LOCK TABLE "user", otp_challenge, "session" IN ACCESS EXCLUSIVE MODE;

DO $check$
BEGIN
  IF EXISTS (SELECT 1 FROM "session" s LEFT JOIN "user" u ON u.id = s.user_id WHERE u.id IS NULL) THEN
    RAISE EXCEPTION 'Orphan session exists; repair session.user_id before ID migration';
  END IF;
END $check$;

CREATE TEMP TABLE _auth_user_ids ON COMMIT DROP AS
  SELECT id AS old_id, row_number() OVER (ORDER BY id)::integer AS new_id FROM "user";
CREATE TEMP TABLE _auth_otp_ids ON COMMIT DROP AS
  SELECT id AS old_id, row_number() OVER (ORDER BY id)::integer AS new_id FROM otp_challenge;
CREATE TEMP TABLE _auth_session_ids ON COMMIT DROP AS
  SELECT id AS old_id, row_number() OVER (ORDER BY id)::integer AS new_id FROM "session";

ALTER TABLE "user" RENAME COLUMN id TO legacy_uuid;
ALTER TABLE "user" ADD COLUMN id integer;
UPDATE "user" u SET id = m.new_id FROM _auth_user_ids m WHERE u.legacy_uuid = m.old_id;
ALTER TABLE "user" DROP CONSTRAINT user_pkey;
ALTER TABLE "user" ALTER COLUMN legacy_uuid DROP NOT NULL;
ALTER TABLE "user" ALTER COLUMN id SET NOT NULL;
ALTER TABLE "user" ADD CONSTRAINT user_pkey PRIMARY KEY (id);
ALTER TABLE "user" ALTER COLUMN id ADD GENERATED BY DEFAULT AS IDENTITY;

ALTER TABLE otp_challenge RENAME COLUMN id TO legacy_uuid;
ALTER TABLE otp_challenge ADD COLUMN id integer;
UPDATE otp_challenge c SET id = m.new_id FROM _auth_otp_ids m WHERE c.legacy_uuid = m.old_id;
ALTER TABLE otp_challenge DROP CONSTRAINT otp_challenge_pkey;
ALTER TABLE otp_challenge ALTER COLUMN legacy_uuid DROP NOT NULL;
ALTER TABLE otp_challenge ALTER COLUMN id SET NOT NULL;
ALTER TABLE otp_challenge ADD CONSTRAINT otp_challenge_pkey PRIMARY KEY (id);
ALTER TABLE otp_challenge ALTER COLUMN id ADD GENERATED BY DEFAULT AS IDENTITY;

ALTER TABLE "session" RENAME COLUMN id TO legacy_uuid;
ALTER TABLE "session" RENAME COLUMN user_id TO legacy_user_uuid;
ALTER TABLE "session" ADD COLUMN id integer;
ALTER TABLE "session" ADD COLUMN user_id integer;
UPDATE "session" s SET id = sm.new_id, user_id = um.new_id
  FROM _auth_session_ids sm, _auth_user_ids um
  WHERE s.legacy_uuid = sm.old_id AND s.legacy_user_uuid = um.old_id;
ALTER TABLE "session" DROP CONSTRAINT refresh_session_pkey;
ALTER TABLE "session" DROP CONSTRAINT refresh_session_user_id_unique;
ALTER TABLE "session" ALTER COLUMN legacy_uuid DROP NOT NULL;
ALTER TABLE "session" ALTER COLUMN legacy_user_uuid DROP NOT NULL;
ALTER TABLE "session" ALTER COLUMN id SET NOT NULL;
ALTER TABLE "session" ALTER COLUMN user_id SET NOT NULL;
ALTER TABLE "session" ADD CONSTRAINT refresh_session_pkey PRIMARY KEY (id);
ALTER TABLE "session" ADD CONSTRAINT refresh_session_user_id_unique UNIQUE (user_id);
ALTER TABLE "session" ALTER COLUMN id ADD GENERATED BY DEFAULT AS IDENTITY;

SELECT setval(pg_get_serial_sequence('"user"', 'id'),
  COALESCE((SELECT max(id) FROM "user"), 1), EXISTS (SELECT 1 FROM "user"));
SELECT setval(pg_get_serial_sequence('otp_challenge', 'id'),
  COALESCE((SELECT max(id) FROM otp_challenge), 1), EXISTS (SELECT 1 FROM otp_challenge));
SELECT setval(pg_get_serial_sequence('"session"', 'id'),
  COALESCE((SELECT max(id) FROM "session"), 1), EXISTS (SELECT 1 FROM "session"));
COMMIT;
~~~

앱을 UUID 버전으로 되돌릴 때만 다음 SQL을 실행한다. 전환 뒤 새로 생성된 행에는 UUID를 부여하고, 세션의 현재 정수 `user_id`를 기준으로 사용자 연결을 복원한다. 롤백 뒤에도 이미 발급한 정수 ID·토큰은 이전 앱과 호환되지 않는다.

~~~sql
BEGIN;
LOCK TABLE "user", otp_challenge, "session" IN ACCESS EXCLUSIVE MODE;
UPDATE "user" SET legacy_uuid = gen_random_uuid() WHERE legacy_uuid IS NULL;
UPDATE otp_challenge SET legacy_uuid = gen_random_uuid() WHERE legacy_uuid IS NULL;
UPDATE "session" SET legacy_uuid = gen_random_uuid() WHERE legacy_uuid IS NULL;
DO $check$
BEGIN
  IF EXISTS (SELECT 1 FROM "session" s LEFT JOIN "user" u ON u.id = s.user_id WHERE u.id IS NULL) THEN
    RAISE EXCEPTION 'session.user_id references a missing user';
  END IF;
END $check$;
UPDATE "session" s SET legacy_user_uuid = u.legacy_uuid
  FROM "user" u WHERE s.user_id = u.id;

ALTER TABLE "session" DROP CONSTRAINT refresh_session_pkey;
ALTER TABLE "session" DROP CONSTRAINT refresh_session_user_id_unique;
ALTER TABLE "session" DROP COLUMN id;
ALTER TABLE "session" DROP COLUMN user_id;
ALTER TABLE "session" RENAME COLUMN legacy_uuid TO id;
ALTER TABLE "session" RENAME COLUMN legacy_user_uuid TO user_id;
ALTER TABLE "session" ALTER COLUMN id SET NOT NULL;
ALTER TABLE "session" ALTER COLUMN user_id SET NOT NULL;
ALTER TABLE "session" ADD CONSTRAINT refresh_session_pkey PRIMARY KEY (id);
ALTER TABLE "session" ADD CONSTRAINT refresh_session_user_id_unique UNIQUE (user_id);

ALTER TABLE otp_challenge DROP CONSTRAINT otp_challenge_pkey;
ALTER TABLE otp_challenge DROP COLUMN id;
ALTER TABLE otp_challenge RENAME COLUMN legacy_uuid TO id;
ALTER TABLE otp_challenge ALTER COLUMN id SET NOT NULL;
ALTER TABLE otp_challenge ADD CONSTRAINT otp_challenge_pkey PRIMARY KEY (id);

ALTER TABLE "user" DROP CONSTRAINT user_pkey;
ALTER TABLE "user" DROP COLUMN id;
ALTER TABLE "user" RENAME COLUMN legacy_uuid TO id;
ALTER TABLE "user" ALTER COLUMN id SET NOT NULL;
ALTER TABLE "user" ADD CONSTRAINT user_pkey PRIMARY KEY (id);
COMMIT;
~~~
